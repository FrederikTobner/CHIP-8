51#define GENERATE_DYNAMIC_ARRAY_TYPE(TYPE) \
52 typedef struct { \
53 size_t allocated; \
54 size_t used; \
55 TYPE * data; \
56 } TYPE##_dynamic_array_t;
57
58#define GENERATE_DYNAMIC_ARRAY_PROTOTYPES(TYPE) \
59 bool TYPE##_dynamic_array_init(TYPE##_dynamic_array_t * array, size_t init_size, size_t init_used); \
60 void TYPE##_dynamic_array_free(TYPE##_dynamic_array_t * array); \
61 bool TYPE##_dynamic_array_resize(TYPE##_dynamic_array_t * array, size_t new_size); \
62 bool TYPE##_dynamic_array_write(TYPE##_dynamic_array_t * array, TYPE val);

64#define GENERATE_DYNAMIC_ARRAY_FUNCTIONS(TYPE) \
65 bool TYPE##_dynamic_array_init(TYPE##_dynamic_array_t * array, size_t init_size, size_t init_used) { \
66 assert(init_size >= init_used); \
67 init_size = MAX(init_size, MIN_ARRAY_SIZE); \
68 array->data = CHECKED_MALLOC_USING_TYPE(init_size, *array->data); \
69 array->allocated = (array->data) ? init_size : 0; \
70 array->used = (array->data) ? init_used : 0; \
71 return !!array->data; \
72 } \
73 \
74 void TYPE##_dynamic_array_free(TYPE##_dynamic_array_t * array) { \
75 free(array->data); \
76 array->data = NULL; \
77 array->allocated = array->used = 0; \
78 } \
79 \
80 bool TYPE##_dynamic_array_resize(TYPE##_dynamic_array_t * array, size_t new_size) { \
81 size_t alloc_size = MAX(new_size, MIN_ARRAY_SIZE); \
82 TYPE * new_data = CHECKED_REALLOC_USING_SIZE(array->data, alloc_size, *array->data); \
83 if (!new_data) \
84 return false; \
85 array->data = new_data; \
86 array->allocated = alloc_size; \
87 array->used = MIN(array->used, new_size); \
88 return true; \
89 } \
90 \
91 bool TYPE##_dynamic_array_write(TYPE##_dynamic_array_t * array, TYPE val) { \
92 if (array->used == array->allocated) { \
93 if (AT_MAX_LENGTH_USING_TYPE(array->allocated, *array->data)) \
94 return false; \
95 size_t new_size = CAPPED_DPL_USING_SIZE(array->allocated, *array->data); \
96 int resize_success = TYPE##_dynamic_array_resize(array, new_size); \
97 if (!resize_success) \
98 return false; \
99 } \
100 array->data[array->used++] = val; \
101 return true; \
102 }
